#!/usr/bin/env bash
set -Eeuo pipefail

# ToolAlign Core-7 full experiment for a fresh Vast.ai Ubuntu instance.
# Run with: bash toolalign.txt

WORKSPACE="${WORKSPACE:-/workspace}"
STEERING_DIR="${WORKSPACE}/jlens-causal-steering"
TOOLALIGN_DIR="${WORKSPACE}/ToolAlignBench"
VENV_DIR="${STEERING_DIR}/.venv"

STEERING_REPO="https://github.com/jinuk0211/jlens-causal-steering.git"
STEERING_BRANCH="agent/jlens-thought-steering"
STEERING_COMMIT="bfc6256140e0e9758466f2dff069475c7b7eda69"

TOOLALIGN_REPO="https://github.com/jinuk0211/ToolAlignBench.git"
TOOLALIGN_BRANCH="agent/toolalign-full-trajectory-jlens"
TOOLALIGN_COMMIT="079b6bf7b5cc4ac952aa660acfc91ae1ee92ca91"

ALIGNED_MODEL="meta-llama/Llama-3.1-8B-Instruct"
ALIGNED_REVISION="0e9e39f249a16976918f6564b8830bc894c89659"
ABLITERATED_MODEL="mlabonne/Meta-Llama-3.1-8B-Instruct-abliterated"
ABLITERATED_REVISION="368c8ed94ce4c986e7b9ca5c159651ef753908ce"

export HF_HOME="${HF_HOME:-${WORKSPACE}/.cache/huggingface}"
export TOKENIZERS_PARALLELISM=false
export HF_HUB_DISABLE_TELEMETRY=1
export PYTORCH_CUDA_ALLOC_CONF="${PYTORCH_CUDA_ALLOC_CONF:-expandable_segments:True}"
export CUDA_VISIBLE_DEVICES="${CUDA_VISIBLE_DEVICES:-0}"

: "${HF_TOKEN:?Set HF_TOKEN to a Hugging Face token with Llama 3.1 access}"

ensure_host_tools() {
  if command -v git >/dev/null && command -v curl >/dev/null; then
    return
  fi
  if ! command -v apt-get >/dev/null; then
    echo "git and curl are required" >&2
    exit 1
  fi
  apt-get update
  apt-get install -y git curl ca-certificates
}

checkout_required_repo() {
  local repo_url="$1"
  local branch="$2"
  local commit="$3"
  local destination="$4"

  if [[ -e "${destination}" && ! -d "${destination}/.git" ]]; then
    echo "${destination} exists but is not a Git repository" >&2
    exit 1
  fi
  if [[ ! -d "${destination}/.git" ]]; then
    git clone --branch "${branch}" --single-branch "${repo_url}" "${destination}"
  else
    git -C "${destination}" fetch origin \
      "+refs/heads/${branch}:refs/remotes/origin/${branch}"
    git -C "${destination}" checkout "${branch}"
    git -C "${destination}" merge --ff-only "origin/${branch}"
  fi
  if ! git -C "${destination}" merge-base --is-ancestor "${commit}" HEAD; then
    echo "${destination} does not contain required commit ${commit}" >&2
    exit 1
  fi
  echo "$(basename "${destination}"): $(git -C "${destination}" rev-parse HEAD)"
}

ensure_uv() {
  if ! command -v uv >/dev/null; then
    curl -LsSf https://astral.sh/uv/install.sh | sh
    export PATH="${HOME}/.local/bin:${PATH}"
  fi
  command -v uv >/dev/null
}

ensure_host_tools
mkdir -p "${WORKSPACE}" "${HF_HOME}"
checkout_required_repo \
  "${STEERING_REPO}" "${STEERING_BRANCH}" "${STEERING_COMMIT}" "${STEERING_DIR}"
checkout_required_repo \
  "${TOOLALIGN_REPO}" "${TOOLALIGN_BRANCH}" "${TOOLALIGN_COMMIT}" "${TOOLALIGN_DIR}"

ensure_uv
uv python install 3.12
if [[ ! -x "${VENV_DIR}/bin/python" ]]; then
  uv venv --python 3.12 "${VENV_DIR}"
fi
source "${VENV_DIR}/bin/activate"
uv pip install -e "${STEERING_DIR}"

export ALIGNED_MODEL ALIGNED_REVISION ABLITERATED_MODEL ABLITERATED_REVISION
python - <<'PY'
import json
import os
from pathlib import Path

config_dir = Path(os.environ["VIRTUAL_ENV"]).parent / "configs"
for path in sorted(config_dir.glob("toolalign_*_llama8b.json")):
    config = json.loads(path.read_text(encoding="utf-8"))
    models = config["models"]
    assert models["aligned"]["model_id"] == os.environ["ALIGNED_MODEL"], path
    assert models["aligned"]["model_revision"] == os.environ["ALIGNED_REVISION"], path
    assert models["abliterated"]["model_id"] == os.environ["ABLITERATED_MODEL"], path
    assert models["abliterated"]["model_revision"] == os.environ["ABLITERATED_REVISION"], path
    assert models["aligned"]["dtype"] == "bfloat16", path
    assert models["abliterated"]["dtype"] == "bfloat16", path
print("ToolAlign model configs verified")
PY

python - <<'PY'
import torch

if not torch.cuda.is_available():
    raise SystemExit("CUDA is not available")
if not torch.cuda.is_bf16_supported():
    raise SystemExit("The selected GPU does not support bfloat16")
print("torch:", torch.__version__)
print("GPU:", torch.cuda.get_device_name(0))
print("BF16: supported")
PY

if [[ "${SKIP_MODEL_DOWNLOAD:-0}" != "1" ]]; then
  hf download "${ALIGNED_MODEL}" --revision "${ALIGNED_REVISION}"
  hf download "${ABLITERATED_MODEL}" --revision "${ABLITERATED_REVISION}"
fi

cd "${STEERING_DIR}"
mkdir -p logs outputs

METHODS=(caa cast mera sadi iti austeer)
for method in "${METHODS[@]}"; do
  echo "===== ToolAlign ${method}: full aligned + abliterated ====="
  bash scripts/remote_toolalign_core7.sh "${method}" full both \
    2>&1 | tee "logs/toolalign-${method}.log"
done

python scripts/summarize_toolalign_core7.py \
  --config-dir configs \
  --output-dir outputs/toolalign-core7-summary

# J-Servo is separate from Core-7. Enable with RUN_JSERVO=1 bash toolalign.txt.
if [[ "${RUN_JSERVO:-0}" == "1" ]]; then
  JSERVO_CONFIG="configs/toolalign_jservo_llama8b.json"
  python scripts/run_toolalign_jservo.py validate "${JSERVO_CONFIG}"
  for role in aligned abliterated; do
    python scripts/run_toolalign_jservo.py baseline "${JSERVO_CONFIG}" --role "${role}"
  done
  python scripts/run_toolalign_jservo.py pairs "${JSERVO_CONFIG}"
  for role in aligned abliterated; do
    python scripts/run_toolalign_jservo.py extract "${JSERVO_CONFIG}" --role "${role}"
    python scripts/run_toolalign_jservo.py sweep "${JSERVO_CONFIG}" --role "${role}"
    python scripts/run_toolalign_jservo.py analyze "${JSERVO_CONFIG}" --role "${role}"
  done
fi

echo "ToolAlign Core-7 complete"
echo "Summary JSON: ${STEERING_DIR}/outputs/toolalign-core7-summary/paired-summary.json"
echo "Summary CSV:  ${STEERING_DIR}/outputs/toolalign-core7-summary/paired-summary.csv"
echo "Per-method logs: ${STEERING_DIR}/logs/"
